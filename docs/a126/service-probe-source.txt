import {beforeEach,expect,it,vi} from 'vitest';
import {Prisma} from '@prisma/client';
const m=vi.hoisted(()=>({actor:vi.fn(),company:vi.fn(),prices:vi.fn(),orders:vi.fn(),users:vi.fn(),subscriptions:vi.fn(),transaction:vi.fn(),raw:vi.fn(),pending:vi.fn(),create:vi.fn(),audit:vi.fn()}));
vi.mock('@/lib/auth/authorization',()=>({AuthorizationError:class extends Error{},requireUser:m.actor,requireUserForMutation:m.actor,requireGlobalSuperAdmin:vi.fn(),requireGlobalSuperAdminForMutation:vi.fn()}));
vi.mock('@/lib/db',()=>({db:{company:{findUnique:m.company},billingPrice:{findMany:m.prices},billingOrder:{findMany:m.orders},user:{findMany:m.users},companySubscription:{findMany:m.subscriptions},$transaction:m.transaction}}));
import {billingDashboard,createAccountPackageOrder} from '@/lib/billing/service';
beforeEach(()=>{vi.clearAllMocks();m.actor.mockResolvedValue({id:'admin',companyId:'company',role:'ACCOUNT_USER',isActive:true,accountRole:'ACCOUNT_ADMIN',accountAccessActive:true,salesAccessActive:false,salesRole:null});m.company.mockResolvedValue({id:'company',productEdition:'SALESPUNCH360_ACCOUNT',subscriptionStatus:'INACTIVE'});m.orders.mockResolvedValue([]);m.users.mockResolvedValue([]);m.subscriptions.mockResolvedValue([]);m.prices.mockResolvedValue([{role:'ACCOUNT_PACKAGE',period:'SIX_MONTH',amount:new Prisma.Decimal(900)},{role:'ACCOUNT_PACKAGE',period:'YEARLY',amount:new Prisma.Decimal(1500)}]);m.pending.mockResolvedValue([]);m.raw.mockResolvedValue([]);m.create.mockImplementation(({data}:any)=>({id:'synthetic-order',...data}));m.transaction.mockImplementation((fn:any)=>fn({$queryRaw:m.raw,company:{findUnique:m.company},billingOrder:{findMany:m.pending,create:m.create},billingAuditEvent:{create:m.audit}}))});
it('reproduces Account-only screen fallback rate differing from authoritative order rate',async()=>{const dashboard=await billingDashboard();expect(dashboard.account.sixMonthUnitPrice).toBe(400);expect(m.prices).not.toHaveBeenCalled();const order=await createAccountPackageOrder(1,'SIX_MONTH');expect(order.accountPackageUnitPrice.toString()).toBe('900');expect(order.totalAmount.toString()).toBe('900')});
it('Account-only service rejects fractional quantities without creating an order',async()=>{await expect(createAccountPackageOrder(1.5,'SIX_MONTH')).rejects.toThrow('INVALID_ACCOUNT_PACKAGE_QUANTITY');expect(m.transaction).not.toHaveBeenCalled()});

it('reproduces reuse of an unrelated pending package/period selection',async()=>{m.pending.mockResolvedValue([{id:'old-pending',provider:'ACCOUNT_PACKAGE',accountPackages:1,billingPeriod:'YEARLY',accountPackageUnitPrice:new Prisma.Decimal(700),totalAmount:new Prisma.Decimal(700)}]);const order=await createAccountPackageOrder(3,'SIX_MONTH');expect(order.id).toBe('old-pending');expect(order.accountPackages).toBe(1);expect(order.billingPeriod).toBe('YEARLY');expect(m.create).not.toHaveBeenCalled()});
it('identical pending intent is reused without creating duplicates',async()=>{m.pending.mockResolvedValue([{id:'same-pending',provider:'ACCOUNT_PACKAGE',accountPackages:1,billingPeriod:'SIX_MONTH',accountPackageUnitPrice:new Prisma.Decimal(900),totalAmount:new Prisma.Decimal(900)}]);const order=await createAccountPackageOrder(1,'SIX_MONTH');expect(order.id).toBe('same-pending');expect(m.create).not.toHaveBeenCalled()});
it('rejects an Account order for a company outside the edition',async()=>{m.company.mockResolvedValue({id:'company',productEdition:'SALESPUNCH360'});await expect(createAccountPackageOrder(1,'SIX_MONTH')).rejects.toThrow();expect(m.transaction).not.toHaveBeenCalled()});
it('rejects an unsupported period before starting a transaction',async()=>{await expect(createAccountPackageOrder(1,'MONTHLY' as never)).rejects.toThrow('INVALID_BILLING_PERIOD');expect(m.transaction).not.toHaveBeenCalled()});
