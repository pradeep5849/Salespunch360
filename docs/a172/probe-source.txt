import {readFileSync} from 'node:fs';
import {describe,it,expect} from 'vitest';
import {workCategorySchema,workPackageSchema} from './validation';
describe('A172 work-category audit',()=>{
 it('category cards lack requested Add Work and category navigation',()=>{const web=readFileSync('src/app/workspace/account/[master]/page.tsx','utf8');const cards=web.slice(web.indexOf('{rows.map((x)'),web.indexOf('<nav>',web.indexOf('{rows.map((x)')));expect(cards).not.toMatch(/Add Work|work-packages|workCategoryId|href=/);});
 it('work create requires category identity and rejects missing category',()=>{expect(()=>workPackageSchema.parse({name:'Wall painting',sellingRate:'100'})).toThrow();expect(workPackageSchema.parse({name:'Wall painting',workCategoryId:'11111111-1111-4111-8111-111111111111',sellingRate:'100'}).workCategoryId).toBeTruthy();});
 it('category schema trims name and rejects blank',()=>{expect(workCategorySchema.parse({name:' Painting '})).toEqual({name:'Painting'});expect(()=>workCategorySchema.parse({name:' '})).toThrow();});
 it('work writer validates tenant and active category before creation',()=>{const service=readFileSync('src/lib/account/service.ts','utf8');const work=service.slice(service.indexOf('export async function createWorkPackage'),service.indexOf('export async function createCustomField'));expect(work).toContain('where: { id: d.workCategoryId, companyId: a.companyId!, isActive: true }');expect(work).toContain('throw new Error("INVALID_WORK_CATEGORY")');});
});
