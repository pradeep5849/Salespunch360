import {beforeEach,expect,it,vi} from 'vitest';
import {readFileSync} from 'node:fs';
const m=vi.hoisted(()=>({actor:vi.fn(),customers:vi.fn(),scope:vi.fn(),branch:vi.fn(),user:vi.fn(),findCustomer:vi.fn(),update:vi.fn(),findLead:vi.fn(),createLead:vi.fn(),activity:vi.fn()}));
vi.mock('@/lib/auth/authorization',()=>({requirePermission:m.actor,requirePermissionForMutation:m.actor}));
vi.mock('@/lib/branches/operational-scope',()=>({operationalBranchContext:m.scope,resolveOperationalWriteBranch:m.branch}));
vi.mock('@/lib/db',()=>({db:{customer:{findMany:m.customers},$transaction:async(fn:any)=>fn({user:{findFirst:m.user},customer:{findFirst:m.findCustomer,update:m.update},lead:{findFirst:m.findLead,create:m.createLead},leadActivity:{create:m.activity}})}}));
import {searchCustomers,searchCustomersForCompany,assignCustomer} from '@/lib/customers/service';
const cid='11111111-1111-4111-8111-111111111111',aid='22222222-2222-4222-8222-222222222222',bid='33333333-3333-4333-8333-333333333333';
beforeEach(()=>{vi.clearAllMocks();m.actor.mockResolvedValue({id:'admin',companyId:'company-A',salesRole:'ADMIN'});m.scope.mockResolvedValue({branchId:{in:['allowed']},branchIds:['allowed']});m.branch.mockResolvedValue('allowed');m.customers.mockResolvedValue([]);m.user.mockImplementation(async({where}:any)=>({id:where.id}));m.findCustomer.mockResolvedValue({id:cid,branchId:'allowed',name:'Customer',phone:'+12025550110'});m.findLead.mockResolvedValue(null);m.createLead.mockResolvedValue({id:'synthetic-lead'})});
it('web default list is bounded to 100 with stable ordering',async()=>{await searchCustomersForCompany('company-A','',null,['allowed']);expect(m.customers.mock.calls[0][0]).toMatchObject({take:100,skip:0,where:{companyId:'company-A',assignedUserId:null,branchId:{in:['allowed']}},orderBy:[{name:'asc'},{id:'asc'}]})});
it('field Sales list binds actor assignment and authorized branches',async()=>{m.actor.mockResolvedValue({id:'sales',companyId:'company-A',salesRole:'SALES'});await searchCustomers(' name ');expect(m.customers.mock.calls[0][0].where).toMatchObject({companyId:'company-A',assignedUserId:'sales',branchId:{in:['allowed']}})});
it('web page offers no search or page inputs to bounded list',()=>{const p=readFileSync('src/app/workspace/customers/page.tsx','utf8');const form=readFileSync('src/app/workspace/customers/customer-manager.tsx','utf8');expect(p).not.toContain('searchParams');expect(form).not.toContain('Load more');expect(form).not.toContain('name="q"')});
it('native Admin queue has hard cap without paging controls',()=>{const service=readFileSync('src/lib/mobile/customers.ts','utf8');const ui=readFileSync('android/app/src/main/java/com/salespunch360/mobile/ui/CustomerAdminScreen.kt','utf8');expect(service).toContain('take:100');expect(ui).not.toContain('Load more');expect(ui).not.toContain('Next')});
it('web assign form has no busy state or returned result display',()=>{const s=readFileSync('src/app/workspace/customers/customer-manager.tsx','utf8');expect(s).toContain('action={assignCustomerAction}');expect(s).not.toContain('useActionState');expect(s).not.toContain('useFormStatus');expect(s).not.toContain('disabled=');expect(s).not.toContain('role="status"')});
it('reproduces both contenders writing after identical unassigned snapshots',async()=>{let arrived=0;let release!:()=>void;const barrier=new Promise<void>(r=>release=r);m.findCustomer.mockImplementation(async()=>{arrived++;if(arrived===2)release();await barrier;return{id:cid,branchId:'allowed',name:'Customer',phone:'+12025550110'}});await Promise.all([assignCustomer({customerId:cid,assignedUserId:aid}),assignCustomer({customerId:cid,assignedUserId:bid})]);expect(m.update).toHaveBeenCalledTimes(2);expect(m.update.mock.calls.map(([q])=>q.where)).toEqual([{id:cid},{id:cid}]);expect(m.createLead).toHaveBeenCalledTimes(2)});
it('outside-scope or already assigned customer rejects before update',async()=>{m.findCustomer.mockResolvedValue(null);await expect(assignCustomer({customerId:cid,assignedUserId:aid})).rejects.toThrow('CUSTOMER_NOT_FOUND');expect(m.update).not.toHaveBeenCalled();expect(m.findCustomer.mock.calls[0][0].where).toMatchObject({companyId:'company-A',assignedUserId:null,branchId:{in:['allowed']}})});
it('mobile assignment shares unconditional final customer update',()=>{const s=readFileSync('src/lib/mobile/customers.ts','utf8');expect(s).toContain('await tx.customer.update({where:{id:customer.id},data:{assignedUserId:assignee.id}})')});
