import {beforeEach,expect,it,vi} from 'vitest';
import {renderToStaticMarkup} from 'react-dom/server';
const m=vi.hoisted(()=>({order:vi.fn(),redirect:vi.fn((path:string)=>{throw new Error('REDIRECT:'+path)})}));
vi.mock('@/lib/billing/service',()=>({billingOrderForCurrentCompany:m.order}));
vi.mock('next/navigation',()=>({redirect:m.redirect}));
import Page from '@/app/workspace/billing/pending/page';
import {ACCOUNT_PACKAGE_ORDER_PROVIDER} from '@/lib/billing/account-package';
import {PLUS_ORDER_PROVIDER} from '@/lib/billing/combined-order';
const base={id:'12345678-abcd-4000-8000-000000000001',billingPeriod:'SIX_MONTH',adminSeats:1,managerSeats:2,salesSeats:3,accountPackages:0,currency:'INR',totalAmount:100.50,status:'PENDING',provider:'MANUAL',createdAt:new Date('2026-11-01'),expiresAt:new Date('2026-11-02')};
const render=async(q:any={order:base.id})=>renderToStaticMarkup(await Page({searchParams:Promise.resolve(q)}));
beforeEach(()=>{vi.clearAllMocks();m.order.mockResolvedValue({...base})});
it('missing ID redirects before loading an order',async()=>{await expect(render({})).rejects.toThrow('REDIRECT:/workspace/billing');expect(m.order).not.toHaveBeenCalled()});
it('uses the first query value and displays pending contact without mutation controls',async()=>{const s=await render({order:[base.id,'other']});expect(m.order).toHaveBeenCalledWith(base.id);expect(s).toContain('Contact on WhatsApp');expect(s).not.toContain('<form');expect(s).toContain('rel="noreferrer"')});
it.each(['PAID','CANCELLED','EXPIRED'])('does not invite payment for terminal %s status',async(status)=>{m.order.mockResolvedValue({...base,status});const s=await render();expect(s).toContain('This order is no longer pending');expect(s).not.toContain('Contact on WhatsApp')});
it('reproduces expired pending order presented as successful with payment invitation',async()=>{const s=await render();expect(base.expiresAt.getTime()).toBeLessThan(Date.now());expect(s).toContain('Your Order Has Been Placed Successfully');expect(s).toContain('Contact on WhatsApp');expect(s).not.toContain('2026-11-02')});
it('reproduces paise rounded in both total and outbound payment message',async()=>{const s=await render();expect(s).toContain('₹101');const href=s.match(/href="(https:\/\/wa.me\/[^\"]+)"/)![1];expect(decodeURIComponent(href)).toContain('Total: ₹101');expect(s).not.toContain('100.50')});
it('renders Account and Plus package counts',async()=>{m.order.mockResolvedValue({...base,provider:ACCOUNT_PACKAGE_ORDER_PROVIDER,accountPackages:2});expect(await render()).toContain('2 Account packages');m.order.mockResolvedValue({...base,provider:PLUS_ORDER_PROVIDER,accountPackages:2});expect(await render()).toContain('1 Admin · 2 Manager · 3 Sales · 2 Account packages')});
it('has no local recovery for inaccessible or missing order',async()=>{m.order.mockRejectedValue(new Error('NOT_FOUND'));await expect(render()).rejects.toThrow('NOT_FOUND')});
it('reproduces missing Telecaller detail even when provided by a future loader',async()=>{m.order.mockResolvedValue({...base,telecallerSeats:4});const s=await render();expect(s.toLowerCase()).not.toContain('telecaller')});
