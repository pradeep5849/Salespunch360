import {beforeEach,expect,it,vi} from 'vitest';
import {renderToStaticMarkup} from 'react-dom/server';
import {readFileSync} from 'node:fs';
const m=vi.hoisted(()=>({dashboard:vi.fn(),tele:vi.fn(),redirect:vi.fn((s:string)=>{throw new Error('REDIRECT:'+s)})}));
vi.mock('@/lib/billing/service',()=>({billingDashboard:m.dashboard}));
vi.mock('@/lib/billing/telecaller',()=>({getTelecallerBillingOverview:m.tele}));
vi.mock('next/navigation',()=>({redirect:m.redirect}));
import Renewal from '@/app/workspace/billing/renewal/page';
import Subscription from '@/app/workspace/billing/subscription/page';
import Telecaller from '@/app/workspace/billing/telecaller/page';
import {loadBillingPageData} from '@/app/workspace/billing/billing-page-data';
const account=()=>({packageCount:2,subscriptions:[],sixMonthUnitPrice:400,yearlyUnitPrice:700,usage:{ACCOUNT_ADMIN:1,ACCOUNTANT:1,PROJECT_MANAGER:0,DATA_ENTRY:0},limits:{ACCOUNT_ADMIN:2,ACCOUNTANT:2,PROJECT_MANAGER:2,DATA_ENTRY:2}});
const ent=()=>({paidActive:true,trialActive:false,state:'PAID',subscription:{endsAt:new Date('2027-01-01'),billingPeriod:'YEARLY'},adminUsage:1,managerUsage:1,salesUsage:2,adminLimit:2,managerLimit:2,salesLimit:5});
const context=(patch:any={})=>({company:{productEdition:'SALESPUNCH360',teamStructure:'MANAGERS_AND_SALES',subscriptionStatus:'ACTIVE',trialEndsAt:new Date('2026-01-01')},hasSales:true,hasAccount:false,entitlement:ent(),prices:[],activeEmployees:[],account:account(),...patch});
beforeEach(()=>{vi.clearAllMocks();m.dashboard.mockResolvedValue(context());m.tele.mockResolvedValue({used:1,limit:3,prices:{SIX_MONTH:600,YEARLY:1000}})});
it('A131 paid Sales renewal submits RENEW and current seat capacity',async()=>{const s=renderToStaticMarkup(await Renewal());expect(s).toContain('Sales Renewal');expect(s).toContain('name="purchaseMode" value="RENEW"');expect(s).toContain('min="0" max="10000" name="salesSeats" value="5"');expect(s).toContain('min="1" max="10000" name="telecallerSeats" value="3"')});
it('A131 inactive Sales uses active usage and NEW intent',async()=>{m.dashboard.mockResolvedValue(context({entitlement:{...ent(),paidActive:false,subscription:null}}));const s=renderToStaticMarkup(await Renewal());expect(s).toContain('name="purchaseMode" value="NEW"');expect(s).toContain('min="0" max="10000" name="salesSeats" value="2"')});
it('A131 Plus includes Account package controls in shared renewal',async()=>{m.dashboard.mockResolvedValue(context({company:{...context().company,productEdition:'SALESPUNCH360_PLUS'},hasAccount:true}));const s=renderToStaticMarkup(await Renewal());expect(s).toContain('Plus Renewal');expect(s).toContain('name="accountPackages"')});
it('A131 reproduces Account-only checkout entry contrary to owner purchase policy',async()=>{m.dashboard.mockResolvedValue(context({hasSales:false,hasAccount:true,entitlement:null}));const s=renderToStaticMarkup(await Renewal());expect(s).toContain('Account Renewal');expect(s).toContain('action="/workspace/billing/checkout/account"');expect(s).toContain('Continue to Checkout');expect(m.tele).not.toHaveBeenCalled()});
it('A131 denied dashboard never renders renewal',async()=>{m.dashboard.mockRejectedValue(new Error('Not authorized'));await expect(Renewal()).rejects.toThrow('Not authorized')});
it('A132 renders Sales and Telecaller limits with no purchase form',async()=>{const s=renderToStaticMarkup(await Subscription());expect(s).toContain('Sales Subscription');expect(s).toContain('3');expect(s).toContain('Telecaller');expect(s).not.toContain('<form')});
it('A132 Account-only renders role usage without Sales data fetch',async()=>{m.dashboard.mockResolvedValue(context({hasSales:false,hasAccount:true,entitlement:null}));const s=renderToStaticMarkup(await Subscription());expect(s).toContain('Account Subscription');expect(s).toContain('1 / 2');expect(s).not.toContain('Sales Subscription');expect(m.tele).not.toHaveBeenCalled()});
it('A132 Plus renders both product summaries',async()=>{m.dashboard.mockResolvedValue(context({hasAccount:true}));const s=renderToStaticMarkup(await Subscription());expect(s).toContain('Sales Subscription');expect(s).toContain('Account Subscription')});
it('A132 reproduces ACTIVE display despite authoritative restricted entitlement',async()=>{m.dashboard.mockResolvedValue(context({entitlement:{...ent(),paidActive:false,trialActive:false,state:'RESTRICTED',subscription:null}}));const d=await loadBillingPageData();expect(d.salesStatus).toBe('ACTIVE');const s=renderToStaticMarkup(await Subscription());expect(s).toContain('<strong>ACTIVE</strong>')});
it('A132 reports trial when effective entitlement is trial',async()=>{m.dashboard.mockResolvedValue(context({entitlement:{...ent(),paidActive:false,trialActive:true,subscription:null}}));expect((await loadBillingPageData()).salesStatus).toBe('TRIAL')});
it('A132 paid Account summary uses latest end although aggregate includes earlier-expiring packages',async()=>{const a=account();a.subscriptions=[{endsAt:new Date('2027-12-01'),accountPackages:1},{endsAt:new Date('2027-01-01'),accountPackages:1}] as any; m.dashboard.mockResolvedValue(context({hasSales:false,hasAccount:true,entitlement:null,account:a}));const s=renderToStaticMarkup(await Subscription());expect(s).toContain('1/12/2027');expect(s).not.toContain('1/1/2027');expect(s).toContain('Account packages')});
it('A133 legacy destination points to nonexistent hub anchor',()=>{expect(()=>Telecaller()).toThrow('REDIRECT:/workspace/billing#sales-add-team');const hub=readFileSync('src/app/workspace/billing/page.tsx','utf8');expect(hub).not.toContain('id="sales-add-team"');expect(hub).not.toContain('/workspace/billing/add-sales-team')});
