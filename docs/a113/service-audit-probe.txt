import ExcelJS from "exceljs";
import {beforeEach,it,expect,vi} from "vitest";
const m=vi.hoisted(()=>({customers:vi.fn(),vendors:vi.fn(),items:vi.fn(),projects:vi.fn(),ledgers:vi.fn(),permission:vi.fn(),modules:vi.fn()}));
vi.mock("@/lib/db",()=>({db:{customer:{findMany:m.customers},vendor:{findMany:m.vendors},accountProduct:{findMany:m.items},project:{findMany:m.projects},ledgerAccount:{findMany:m.ledgers}}}));
vi.mock("@/lib/auth/authorization",()=>({requirePermission:m.permission}));
vi.mock("@/lib/account/modules",()=>({enabledModulesForCompany:m.modules}));
import {exportAccountDataForActor,exportAccountData} from "./exports";
import {parseCsv,previewAccountImportForActor} from "./imports";
const actor={id:"synthetic",companyId:"company-a",accountRole:"ACCOUNTANT",branchAccessScope:"SELECTED_BRANCHES",branchIds:["branch-a"]};
beforeEach(()=>{vi.clearAllMocks();m.customers.mockResolvedValue([{name:"Customer",phone:"0123"}]);m.items.mockResolvedValue([{name:"Item",code:"A1"}]);m.projects.mockResolvedValue([{name:"Project",projectNumber:"P1"}]);m.permission.mockResolvedValue(actor);m.modules.mockResolvedValue([])});
it("captures a harmless formula-like customer name unchanged in CSV",async()=>{m.customers.mockResolvedValue([{name:"=1+1",phone:"0123"}]);const out=await exportAccountDataForActor(actor,"customers","csv"),rows=parseCsv(new TextDecoder().decode(out.bytes));expect(rows[1][0]).toBe("=1+1");console.log("A113_CSV_FORMULA",JSON.stringify({literalCell:rows[1][0],neutralized:false,spreadsheetExecuted:false}))});
it("confirms the same text is a string, not an XLSX formula object",async()=>{m.customers.mockResolvedValue([{name:"=1+1"}]);const out=await exportAccountDataForActor(actor,"customers","xlsx"),book=new ExcelJS.Workbook();await book.xlsx.load(out.bytes.buffer as ArrayBuffer);expect(book.worksheets[0].getCell("A2").type).toBe(ExcelJS.ValueType.String);expect(book.worksheets[0].getCell("A2").value).toBe("=1+1")});
it("captures the deterministic query cap without an output truncation marker",async()=>{m.customers.mockImplementation(async({take})=>Array.from({length:Math.min(take,10001)},(_,i)=>({name:`customer-${i}`})));const out=await exportAccountDataForActor(actor,"customers","csv"),rows=parseCsv(new TextDecoder().decode(out.bytes));expect(rows.length-1).toBe(10000);expect(m.customers.mock.calls[0][0]).toMatchObject({take:10000});expect(m.customers.mock.calls[0][0].orderBy).toBeUndefined();expect(Object.keys(out).sort()).toEqual(["bytes","contentType"]);console.log("A113_TRUNCATION",JSON.stringify({fixtureRows:10001,exportedRows:rows.length-1,stableQueryOrdering:false,outputWarning:false}))});
it.each([["customers","CUSTOMERS","MISSING_COLUMNS:branch"],["items","ITEMS","MISSING_COLUMNS:unit,category"],["projects","PROJECTS","MISSING_COLUMNS:customer,branch"]])("captures export/import compatibility for %s",async(type,importType,message)=>{const out=await exportAccountDataForActor(actor,type,"csv");await expect(previewAccountImportForActor(actor,{type:importType as never,fileName:`${type}.csv`,bytes:out.bytes})).rejects.toThrow(message);console.log("A113_REIMPORT",JSON.stringify({type,result:message}))});
it("captures an item export without consulting the disabled-Inventory configuration",async()=>{const out=await exportAccountData("items","csv");expect(new TextDecoder().decode(out.bytes)).toContain("Item");expect(m.permission).toHaveBeenCalledWith("ACCOUNT_REPORTS");expect(m.modules).not.toHaveBeenCalled();console.log("A113_ITEM_GATE",JSON.stringify({reportsPermission:true,inventoryConfigurationRead:false,itemDataReturned:true}))});
it("captures format validation occurring after the data query",async()=>{await expect(exportAccountDataForActor(actor,"customers","unsupported")).rejects.toThrow("UNSUPPORTED_EXPORT_FORMAT");expect(m.customers).toHaveBeenCalledOnce()});
it("confirms customer branch filters and project-manager project filters are server-derived",async()=>{await exportAccountDataForActor(actor,"customers","csv");expect(m.customers.mock.calls[0][0].where).toEqual({companyId:"company-a",branchId:{in:["branch-a"]}});await exportAccountDataForActor({...actor,accountRole:"PROJECT_MANAGER"},"projects","csv");expect(m.projects.mock.calls[0][0].where).toEqual({companyId:"company-a",branchId:{in:["branch-a"]},projectManagerId:"synthetic"})});
it("captures empty CSV lacking its dataset headers",async()=>{m.customers.mockResolvedValue([]);const out=await exportAccountDataForActor(actor,"customers","csv");expect(out.bytes.byteLength).toBe(0)});
