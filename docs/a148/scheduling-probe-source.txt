import {beforeEach,expect,it,vi} from 'vitest';
import {readFileSync} from 'node:fs';
const m=vi.hoisted(()=>({find:vi.fn(),create:vi.fn(),next:vi.fn(),update:vi.fn(),raw:vi.fn(),notify:vi.fn(),paid:vi.fn()}));
vi.mock('@/lib/db',()=>({db:{$transaction:async(fn:any)=>fn({$queryRaw:m.raw,lead:{findFirst:m.find,update:m.update},followUpTask:{create:m.create,findFirst:m.next}})}}));
vi.mock('@/lib/billing/entitlement',()=>({assertOperationalWrite:vi.fn()}));
vi.mock('@/lib/billing/telecaller',()=>({assertActiveTelecallerEntitlement:m.paid}));
vi.mock('@/lib/branches/operational-scope',()=>({resolveOperationalWriteBranch:vi.fn(async()=> 'branch')}));
vi.mock('@/lib/sales-notifications/service',()=>({createSalesNotification:m.notify}));
import {createFollowUpTaskForActor} from '@/lib/follow-up-tasks/service';
import {createTelecallerCallFollowUpForActor} from '@/lib/follow-up-tasks/telecaller-assignment';
const actor={id:'aaaaaaaa-aaaa-4aaa-8aaa-aaaaaaaaaaaa',companyId:'cccccccc-cccc-4ccc-8ccc-cccccccccccc',salesRole:'PRIMARY_ADMIN' as const};
const owner='11111111-1111-4111-8111-111111111111',tele='22222222-2222-4222-8222-222222222222';
const input={leadId:'33333333-3333-4333-8333-333333333333',dueDate:'2026-11-20',type:'CALL',notes:'Call customer'};
beforeEach(()=>{vi.clearAllMocks();m.find.mockResolvedValue({id:input.leadId,branchId:'branch',assignedUserId:owner,assignedUser:{id:owner,salesRole:'SALES',isActive:true,salesAccessActive:true,managerId:null,managerType:null}});m.create.mockImplementation(async({data}:any)=>({id:'synthetic-task-'+m.create.mock.calls.length,...data}));m.next.mockResolvedValue({dueDate:new Date('2026-11-20')});m.raw.mockResolvedValue([{id:tele}])});
it('normal lead-owner Call emits an assignment notification',async()=>{const task=await createFollowUpTaskForActor(actor,input);expect(task.assignedUserId).toBe(owner);expect(m.notify).toHaveBeenCalledWith(expect.anything(),expect.objectContaining({recipientUserId:owner,eventType:'FOLLOW_UP_ASSIGNED'}))});
it('Telecaller Call creation writes task but emits no equivalent assignment notification',async()=>{const task=await createTelecallerCallFollowUpForActor(actor,{...input,assignedUserId:tele});expect(task.assignedUserId).toBe(tele);expect(m.paid).toHaveBeenCalledWith(actor.companyId);expect(m.notify).not.toHaveBeenCalled()});
it('repeated identical normal requests create distinct pending tasks and notices',async()=>{const a=await createFollowUpTaskForActor(actor,input),b=await createFollowUpTaskForActor(actor,input);expect(a.id).not.toBe(b.id);expect(m.create).toHaveBeenCalledTimes(2);expect(m.notify).toHaveBeenCalledTimes(2)});
it('repeated identical Telecaller requests also create distinct tasks',async()=>{await createTelecallerCallFollowUpForActor(actor,{...input,assignedUserId:tele});await createTelecallerCallFollowUpForActor(actor,{...input,assignedUserId:tele});expect(m.create).toHaveBeenCalledTimes(2)});
it('invalid calendar date rejects normal request before lead lookup',async()=>{await expect(createFollowUpTaskForActor(actor,{...input,dueDate:'2026-02-30'})).rejects.toThrow('INVALID_DATE');expect(m.find).not.toHaveBeenCalled();expect(m.create).not.toHaveBeenCalled()});
it('invalid Telecaller target or paid entitlement prevents writes',async()=>{m.raw.mockResolvedValue([]);await expect(createTelecallerCallFollowUpForActor(actor,{...input,assignedUserId:tele})).rejects.toThrow('INVALID_ASSIGNMENT');expect(m.create).not.toHaveBeenCalled();m.paid.mockRejectedValueOnce(new Error('TELECALLER_SUBSCRIPTION_REQUIRED'));await expect(createTelecallerCallFollowUpForActor(actor,{...input,assignedUserId:tele})).rejects.toThrow('TELECALLER_SUBSCRIPTION_REQUIRED')});
it('native scheduling dialog is removed immediately after asynchronous request starts',()=>{const s=readFileSync('android/app/src/main/java/com/salespunch360/mobile/ui/LeadsScreen.kt','utf8');expect(s).toContain('vm.followUp(lead,date,notes,type,assignedId);followLead=null');expect(s).toContain('var date by remember{mutableStateOf("")}');const vm=readFileSync('android/app/src/main/java/com/salespunch360/mobile/LeadsViewModel.kt','utf8');expect(vm).toContain('fun followUp(lead:LeadSummary');expect(vm).toContain('followUpMutations.create(lead.id,dueDate,type,notes,assignedUserId)')});
it('web retains submission feedback and only resets fields after confirmed success',()=>{const s=readFileSync('src/app/workspace/leads/follow-up-form.tsx','utf8');expect(s.indexOf('if(!res.ok||!data.ok)')).toBeLessThan(s.indexOf('formRef.current?.reset()'));expect(s).toContain('disabled={pending}');expect(s).toContain('role="alert"')});
