import {expect,it} from 'vitest';
import {readFileSync} from 'node:fs';
import {salesGroups,accountGroups,salesRoleLabel,type DirectoryUser} from '@/app/workspace/employees/directory-policy';
const base:DirectoryUser={id:'tele',name:'Telecaller',email:'t@example.test',isActive:true,salesRole:'SALES',designation:'Telecaller',managerType:null,salesAccessActive:false,accountRole:null,accountAccessActive:false};
it('web correctly keeps active Telecaller in active separate group with Sales access disabled',()=>{const g=salesGroups([base]);expect(g.telecallers).toHaveLength(1);expect(g.active).toHaveLength(1);expect(g.inactive).toHaveLength(0);expect(g.sales).toHaveLength(0);expect(salesRoleLabel(base)).toBe('Telecaller')});
it('native current active helper requires Sales access and role label omits Telecaller designation',()=>{const model=readFileSync('android/app/src/main/java/com/salespunch360/mobile/data/Models.kt','utf8');const screen=readFileSync('android/app/src/main/java/com/salespunch360/mobile/ui/EmployeesScreen.kt','utf8');expect(model).toContain('isActiveEmployee(employee:Employee)=employee.isActive&&employee.salesAccessActive');const start=screen.indexOf('private fun roleLabel');const helper=screen.slice(start,screen.indexOf('\n',start));expect(helper).toContain('MobileRole.SALES->"Sales Employee"');expect(helper).not.toContain('designation')});
it('native EmployeeContext retains Telecaller designation but no dedicated Telecaller seat overview',()=>{const s=readFileSync('src/lib/mobile/employees.ts','utf8');expect(s).toContain('designation:user.designation');expect(s).not.toContain('getTelecallerBillingOverview');expect(s).not.toContain('telecallerLimit')});
it('web directory contains no text search while native directory has search field',()=>{const web=readFileSync('src/app/workspace/employees/page.tsx','utf8');const native=readFileSync('android/app/src/main/java/com/salespunch360/mobile/ui/EmployeesScreen.kt','utf8');expect(web).not.toContain('type="search"');expect(web).not.toContain('name="q"');expect(web).not.toContain('Search employees');expect(native).toContain('Search employees')});
it('web Telecaller activation has direct form with no confirmation/pending/result state',()=>{const s=readFileSync('src/app/workspace/employees/page.tsx','utf8');const start=s.indexOf('function TelecallerGroup');const group=s.slice(start,s.indexOf('export default',start));expect(group).toContain('action={setTelecallerActiveAction}');expect(group).not.toContain('confirm(');expect(group).not.toContain('disabled=');expect(group).not.toContain('role="status"')});
it('shared identity is visible in each authorized domain without duplicating within groups',()=>{const shared={...base,id:'shared',designation:null,salesAccessActive:true,accountRole:'ACCOUNTANT' as const,accountAccessActive:true};expect(salesGroups([shared,shared]).sales).toHaveLength(1);expect(accountGroups([shared,shared]).ACCOUNTANT).toHaveLength(1)});
it('inactive domain access is represented independently across shared identity',()=>{const shared={...base,designation:null,salesAccessActive:true,accountRole:'ACCOUNTANT' as const,accountAccessActive:false};expect(salesGroups([shared]).active).toHaveLength(1);expect(accountGroups([shared]).inactive).toHaveLength(1)});
