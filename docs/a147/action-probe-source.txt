import {beforeEach,expect,it,vi} from 'vitest';
import {readFileSync} from 'node:fs';
const m=vi.hoisted(()=>({edit:vi.fn(),refresh:vi.fn()}));
vi.mock('@/lib/leads/service',()=>({editLead:m.edit,createLead:vi.fn(),createLeadFromVisit:vi.fn()}));
vi.mock('next/cache',()=>({revalidatePath:m.refresh}));
import {saveLead} from '@/app/actions/leads';
import {LeadError} from '@/lib/leads/policy';
const form=()=>{const f=new FormData();f.set('leadId','synthetic');return f};
beforeEach(()=>vi.clearAllMocks());
it('web stale-version edit gives generic details/access message',async()=>{m.edit.mockRejectedValueOnce(new LeadError('STALE'));expect(await saveLead({},form())).toMatchObject({error:'Unable to save the lead. Check its details and your access.'});expect(m.refresh).not.toHaveBeenCalled()});
it('active follow-up conflict gives specific guidance',async()=>{m.edit.mockRejectedValueOnce(new LeadError('ACTIVE_FOLLOW_UP'));expect(await saveLead({},form())).toMatchObject({error:expect.stringContaining('checkout')})});
it('reassignment moves tasks without creating follow-up assignment notifications',()=>{const s=readFileSync('src/lib/leads/service.ts','utf8');const body=s.slice(s.indexOf('export async function editLeadForActor'),s.indexOf('export async function',s.indexOf('export async function editLeadForActor')+20));expect(body).toContain('followUpTask.updateMany');expect(body).not.toContain('createSalesNotification');expect(body).not.toContain('FOLLOW_UP_REASSIGNED')});
it('Android null edit fields fall back to saved values',()=>{const s=readFileSync('src/lib/mobile/leads.ts','utf8');expect(s).toContain('phone:input.phone??current.phone');expect(s).toContain('notes:input.notes??current.notes');const ui=readFileSync('android/app/src/main/java/com/salespunch360/mobile/ui/LeadsScreen.kt','utf8');expect(ui).toContain('phone=phone.ifBlank{null}');expect(ui).toContain('Reload latest version')});
