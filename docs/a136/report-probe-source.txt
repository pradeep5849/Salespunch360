import {beforeEach,expect,it,vi} from 'vitest';
import {readFileSync} from 'node:fs';
const m=vi.hoisted(()=>({visits:vi.fn(),count:vi.fn(),leads:vi.fn(),customers:vi.fn(),scope:vi.fn(),branches:vi.fn()}));
vi.mock('@/lib/db',()=>({db:{customerVisit:{findMany:m.visits,count:m.count},lead:{count:m.leads},customer:{findMany:m.customers}}}));
vi.mock('@/lib/reports/scope',()=>({reportActor:vi.fn(),resolveEmployeeScope:m.scope,reportEmployeeOptions:vi.fn(async()=>[])}));
vi.mock('@/lib/branches/operational-scope',()=>({operationalBranchContext:m.branches}));
import {checkInReport} from '@/lib/reports/check-ins';
const actor={id:'self',companyId:'company-A',salesRole:'SALES' as const,name:'Self'};
const visit=(extra:any={})=>({id:'visit-A',leadId:'existing-lead',visitType:'FOLLOW_UP',checkedInAt:new Date('2026-11-01'),checkedOutAt:new Date('2026-11-01T01:00Z'),checkInLatitude:12,checkInLongitude:77,customer:null,sourceLeads:[],...extra});
beforeEach(()=>{vi.clearAllMocks();m.visits.mockResolvedValue([visit()]);m.count.mockResolvedValue(1);m.leads.mockResolvedValue(0);m.customers.mockResolvedValue([]);m.scope.mockResolvedValue(['self']);m.branches.mockResolvedValue({branchId:{in:['allowed']},branchIds:['allowed']})});
it('report query retains authenticated company, authorized employees and branch restrictions',async()=>{await checkInReport({},false,actor);const q=m.visits.mock.calls[0][0];expect(q.where).toMatchObject({companyId:'company-A',userId:{in:['self']},branchId:{in:['allowed']}});expect(q.orderBy).toEqual([{checkedInAt:'desc'},{id:'desc'}]);expect(q.take).toBeGreaterThan(0)});
it('reproduces existing linked Lead ID overwritten to null for a follow-up visit',async()=>{const r=await checkInReport({},false,actor);expect(r.rows[0].leadId).toBeNull()});
it('newly generated source lead does appear',async()=>{m.visits.mockResolvedValue([visit({sourceLeads:[{id:'generated-lead'}]})]);expect((await checkInReport({},false,actor)).rows[0].leadId).toBe('generated-lead')});
it('Sales default history has no implicit date cutoff',async()=>{await checkInReport({},false,actor);expect(m.visits.mock.calls[0][0].where).not.toHaveProperty('checkedInAt')});
it('explicit day range is applied and page advances bounded queries',async()=>{await checkInReport({start:'2026-11-01',end:'2026-11-01',page:'2'},false,actor);const q=m.visits.mock.calls[0][0];expect(q.where.checkedInAt).toHaveProperty('gte');expect(q.skip).toBe(q.take)});
it('report and field API contain no user visit-delete action',()=>{const web=readFileSync('src/app/workspace/reports/check-ins/page.tsx','utf8');const actions=readFileSync('src/app/actions/visits.ts','utf8');const api=readFileSync('src/app/api/v1/mobile/field/route.ts','utf8');expect(web).not.toMatch(/Delete|deleteVisit/);expect(actions).not.toMatch(/export async function (delete|remove)/);expect(api).not.toContain("body.action==='DELETE'")});
it('report card lacks distinct visit Details action and phone presentation',()=>{const s=readFileSync('src/app/workspace/reports/check-ins/page.tsx','utf8');const card=s.slice(s.indexOf('function CheckInCards'),s.indexOf('function SalesCheckInReport'));expect(card).not.toContain('contactPhone');expect(card).not.toContain('customer.phone');expect(card).not.toContain('Visit Details');expect(card).toContain('Open Lead Details')});
it('Android report card does not present checkout timestamp',()=>{const s=readFileSync('android/app/src/main/java/com/salespunch360/mobile/ui/AdminReportScreens.kt','utf8');const card=s.slice(s.indexOf('val employee=row.adminObjText'),s.indexOf('if(row.adminText("checkoutSentiment")'));expect(card).toContain('adminReportDateTime(it)');expect(card).not.toMatch(/checkedOutAt.*adminReportDateTime/);expect(s).not.toContain('Delete Check-in')});
