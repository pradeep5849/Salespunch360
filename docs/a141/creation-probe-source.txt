import {beforeEach,expect,it,vi} from 'vitest';
import {readFileSync} from 'node:fs';
const m=vi.hoisted(()=>({company:vi.fn(),actor:vi.fn(),create:vi.fn(),lock:vi.fn(),count:vi.fn(),hash:vi.fn(),subs:vi.fn(),branches:vi.fn()}));
vi.mock('@/lib/auth/password',()=>({hashPassword:m.hash}));
vi.mock('@/lib/db',()=>({db:{company:{findUnique:m.company},$transaction:async(fn:any)=>fn({$queryRaw:m.lock,user:{findFirst:m.actor,count:m.count,create:m.create},companySubscription:{findMany:m.subs},userBranchAccess:{deleteMany:m.branches,createMany:vi.fn()}})}}));
import {createAccountUserForActor,createAccountUserSchema} from '@/lib/users/product-user-management';
const actor={id:'admin',companyId:'company-A',role:'ACCOUNT_USER',isActive:true,accountRole:'ACCOUNT_ADMIN',accountAccessActive:true,salesRole:null,salesAccessActive:false};
const company={id:'company-A',productEdition:'SALESPUNCH360_ACCOUNT',teamStructure:'SALES_ONLY',subscriptionStatus:'TRIAL',name:'Company',addressLine1:'Road 1',city:'City',state:'State',postalCode:'12345',country:'India',primaryContactName:'Owner',primaryPhone:'9999999999',contactEmail:'owner@example.test'};
const input={name:' Employee ',email:' EMPLOYEE@example.test ',password:'StrongPassword12',confirmPassword:'StrongPassword12',accountRole:'ACCOUNTANT',branchAccessScope:'ALL_BRANCHES',branchIds:[]};
beforeEach(()=>{vi.clearAllMocks();m.company.mockResolvedValue(company);m.actor.mockResolvedValue({emailVerifiedAt:new Date()});m.count.mockResolvedValue(0);m.subs.mockResolvedValue([]);m.hash.mockResolvedValue('synthetic-hash');m.create.mockImplementation(async({data}:any)=>({id:'synthetic-employee',...data}))});
it('creates only Account access in authenticated company after company lock/readiness',async()=>{const u=await createAccountUserForActor(actor as any,input);expect(u).toMatchObject({companyId:'company-A',name:'Employee',email:'employee@example.test',salesRole:null,salesAccessActive:false,accountRole:'ACCOUNTANT',accountAccessActive:true});expect(m.lock.mock.invocationCallOrder[0]).toBeLessThan(m.create.mock.invocationCallOrder[0]);expect(m.create.mock.calls[0][0].data).not.toHaveProperty('password')});
it('unverified actor cannot create employee',async()=>{m.actor.mockResolvedValue({emailVerifiedAt:null});await expect(createAccountUserForActor(actor as any,input)).rejects.toThrow('EMAIL_VERIFICATION_REQUIRED');expect(m.create).not.toHaveBeenCalled()});
it('incomplete company profile cannot create employee',async()=>{m.company.mockResolvedValue({...company,addressLine1:null});await expect(createAccountUserForActor(actor as any,input)).rejects.toThrow('COMPANY_PROFILE_REQUIRED');expect(m.create).not.toHaveBeenCalled()});
it('expired workspace cannot create employee or hash password',async()=>{m.company.mockResolvedValue({...company,subscriptionStatus:'EXPIRED'});await expect(createAccountUserForActor(actor as any,input)).rejects.toThrow('WORKSPACE_SUSPENDED');expect(m.hash).not.toHaveBeenCalled();expect(m.create).not.toHaveBeenCalled()});
it('rejects invalid role and password confirmation',()=>{expect(createAccountUserSchema.safeParse({...input,accountRole:'OWNER'}).success).toBe(false);expect(createAccountUserSchema.safeParse({...input,confirmPassword:'different'}).success).toBe(false)});
it('rejects client-owned tenant field',()=>{expect(createAccountUserSchema.safeParse({...input,companyId:'foreign'}).success).toBe(false)});
it('native temporary password field lacks masking and password keyboard options',()=>{const s=readFileSync('android/app/src/main/java/com/salespunch360/mobile/ui/account/admin/AccountAdministrationScreen.kt','utf8');const start=s.indexOf('OutlinedTextField(password,');const field=s.slice(start,s.indexOf('\n',start));expect(field).toContain('Temporary password');expect(field).not.toContain('PasswordVisualTransformation');expect(field).not.toContain('visualTransformation');expect(field).not.toContain('KeyboardType.Password')});
it('native create payload fixes All Branches and empty assignments',()=>{const s=readFileSync('android/app/src/main/java/com/salespunch360/mobile/ui/account/admin/AccountAdministrationScreen.kt','utf8');const create=s.slice(s.indexOf('private fun UserView'),s.indexOf('Text("Effective roles and access")'));expect(create).toContain('put("branchAccessScope", "ALL_BRANCHES")');expect(create).toContain('putJsonArray("branchIds") {}');expect(create).not.toContain('SELECTED_BRANCHES')});
it('web exposes selected-branch creation option',()=>{const s=readFileSync('src/app/workspace/employees/account/new/[role]/page.tsx','utf8');expect(s).toContain('value="SELECTED_BRANCHES"');expect(s).toContain('name="confirmPassword"');expect(s).toContain('manageAccountUserForm')});
