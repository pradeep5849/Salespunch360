import {createElement} from "react";
import {renderToStaticMarkup} from "react-dom/server";
import {Prisma} from "@prisma/client";
import {beforeEach,it,expect,vi} from "vitest";
const m=vi.hoisted(()=>({depth:0,trace:[] as {model:string;depth:number}[],reads:new Map<string,unknown[]>(),year:vi.fn(),update:vi.fn(),period:vi.fn(),audit:vi.fn(),query:vi.fn(),execute:vi.fn(),series:vi.fn(),seriesUpdate:vi.fn(),settings:vi.fn(),settingsSave:vi.fn(),branch:vi.fn(),permission:vi.fn(),years:vi.fn()}));
vi.mock("@/lib/db",()=>{const delegates=new Proxy({}, {get:(_,model)=>{
 if(model==="$queryRaw")return m.query;if(model==="$executeRaw")return m.execute;
 if(model==="financialYear")return {findFirst:m.year,update:m.update,findMany:m.years};
 if(model==="accountingPeriodLock")return {upsert:m.period};
 if(model==="accountingAuditEvent")return {create:m.audit};
 if(model==="numberingSeries")return {findFirst:m.series,update:m.seriesUpdate};
 if(model==="accountSettings")return {findUnique:m.settings,upsert:m.settingsSave};
 if(model==="branch")return {findFirst:m.branch,findMany:async()=>{m.trace.push({model:String(model),depth:m.depth});return []}};
 const read=async()=>{m.trace.push({model:String(model),depth:m.depth});return m.reads.get(String(model))??[]};return {findMany:read,groupBy:read};
 }});return {db:new Proxy(delegates,{get:(target,name)=>name==="$transaction"?async(fn:(tx:unknown)=>Promise<unknown>)=>{m.depth++;try{return await fn(delegates)}finally{m.depth--}}:Reflect.get(target,name)})}});
vi.mock("@/lib/auth/authorization",()=>({requirePermission:m.permission,requirePermissionForMutation:m.permission,AuthorizationError:class extends Error{}}));
vi.mock("@/app/actions/account-finalization",()=>({closeYearAction:async()=>{throw new Error("AUDIT_ACTION_MUST_NOT_EXECUTE")}}));
vi.mock("@/components/account/account-shell",()=>({AccountPageHeader:({title,subtitle}:{title:string;subtitle:string})=>createElement("header",null,title,subtitle),AccountCard:({children}:{children:unknown})=>createElement("section",null,children as never)}));
import {closeFinancialYearForActor} from "./utilities";
import {updateTransactionSettingsForActor} from "./settings";
import {DEFAULT_TRANSACTION_PREFERENCES} from "./transaction-settings";
import Page from "@/app/workspace/account/utilities/financial-year/page";
const actor={id:"admin",companyId:"company-a",accountRole:"ACCOUNT_ADMIN",branchAccessScope:"ALL_BRANCHES"};
const year={id:"year-a",companyId:"company-a",name:"FY fixture",status:"OPEN",startDate:new Date("2024-04-01"),endDate:new Date("2025-03-31"),isActive:true,isCurrent:true};
beforeEach(()=>{vi.clearAllMocks();m.depth=0;m.trace=[];m.reads.clear();m.year.mockResolvedValue(year);m.years.mockResolvedValue([year]);m.update.mockImplementation(async({data})=>({...year,...data}));m.permission.mockResolvedValue(actor);m.query.mockResolvedValue([]);m.execute.mockResolvedValue(0);m.series.mockResolvedValue({id:"series-a",nextSequence:42n});m.seriesUpdate.mockResolvedValue({id:"series-a",nextSequence:42n});m.settings.mockResolvedValue({transactionDefaults:{},itemSettings:{}});m.settingsSave.mockResolvedValue({});m.branch.mockResolvedValue({id:"11111111-1111-4111-8111-111111111111",companyId:"company-a"})});
it("rejects non-admin closure before verification reads",async()=>{await expect(closeFinancialYearForActor({...actor,accountRole:"ACCOUNTANT"},{financialYearId:year.id})).rejects.toThrow();expect(m.trace).toEqual([]);expect(m.update).not.toHaveBeenCalled()});
it("captures history-preserving year closure and end-date lock rather than a Start Fresh operation",async()=>{const out=await closeFinancialYearForActor(actor,{financialYearId:year.id});expect(out.status).toBe("CLOSED");expect(m.update).toHaveBeenCalledWith({where:{id:year.id},data:expect.objectContaining({isActive:false,isCurrent:false,status:"CLOSED"})});expect(m.period.mock.calls[0][0].create.lockedThrough).toEqual(year.endDate);expect(m.audit.mock.calls[0][0].data.eventType).toBe("FINANCIAL_YEAR_CLOSED");console.log("A114_CLOSE_KIND",JSON.stringify({kind:"CLOSE_AND_LOCK_EXISTING_YEAR",deletionMethodAvailableInMock:false,backupRequested:false,closingDate:year.endDate.toISOString()}))});
it("blocks closure when the read-only integrity checks report an error",async()=>{m.reads.set("journalEntry",[{journalNumber:"synthetic-bad",lines:[{debit:new Prisma.Decimal(1),credit:new Prisma.Decimal(0)}]}]);await expect(closeFinancialYearForActor(actor,{financialYearId:year.id})).rejects.toThrow("FINANCIAL_YEAR_VERIFICATION_FAILED");expect(m.update).not.toHaveBeenCalled()});
it("requires a reason to close a future-ended year",async()=>{m.year.mockResolvedValue({...year,endDate:new Date("2099-03-31")});await expect(closeFinancialYearForActor(actor,{financialYearId:year.id})).rejects.toThrow("EARLY_CLOSE_REASON_REQUIRED");expect(m.update).not.toHaveBeenCalled()});
it("permits the existing explicit early-close reason policy",async()=>{m.year.mockResolvedValue({...year,endDate:new Date("2099-03-31")});await closeFinancialYearForActor(actor,{financialYearId:year.id,earlyCloseReason:"SYNTHETIC_ONLY early-close reason"});expect(m.update).toHaveBeenCalledOnce()});
it("scopes the target year read to the authorized company",async()=>{await closeFinancialYearForActor(actor,{financialYearId:year.id});expect(m.year).toHaveBeenCalledWith({where:{id:year.id,companyId:"company-a"}})});
it("captures integrity reads outside the close transaction and only a financial-year row lock",async()=>{await closeFinancialYearForActor(actor,{financialYearId:year.id});expect(m.trace.length).toBeGreaterThanOrEqual(11);expect(m.trace.every(x=>x.depth===0)).toBe(true);const sql=m.query.mock.calls.map(call=>Array.from(call[0] as string[]).join(" ")).join("\n");expect(sql).toContain('"financial_years"');expect(sql).not.toContain('"companies"');console.log("A114_VERIFICATION_BOUNDARY",JSON.stringify({verificationReads:m.trace.length,allBeforeTransaction:true,companyPostingLockObserved:false,realConcurrencyExecuted:false}))});
it("captures closed-year replay blocked by a new company-wide integrity failure",async()=>{m.year.mockResolvedValue({...year,status:"CLOSED"});m.reads.set("journalEntry",[{journalNumber:"different-period-fixture",lines:[{debit:new Prisma.Decimal(1),credit:new Prisma.Decimal(0)}]}]);await expect(closeFinancialYearForActor(actor,{financialYearId:year.id})).rejects.toThrow("FINANCIAL_YEAR_VERIFICATION_FAILED");expect(m.year).not.toHaveBeenCalled();console.log("A114_REPLAY",JSON.stringify({targetAlreadyClosed:true,globalCheckPreventsTargetRead:true}))});
it("captures an unvalidated early-close-reason type reaching trim",async()=>{m.year.mockResolvedValue({...year,endDate:new Date("2099-03-31")});await expect(closeFinancialYearForActor(actor,{financialYearId:year.id,earlyCloseReason:7} as never)).rejects.toThrow(/trim/);expect(m.trace.length).toBeGreaterThan(0);expect(m.update).not.toHaveBeenCalled()});
it("captures generic prefix updates accepting reference-disallowed text without restarting the sequence",async()=>{await updateTransactionSettingsForActor(actor as never,{preferences:DEFAULT_TRANSACTION_PREFERENCES,branchId:"11111111-1111-4111-8111-111111111111",prefixes:{SALES_INVOICE:"0#@"}});expect(m.seriesUpdate).toHaveBeenCalledWith({where:{id:"series-a"},data:{prefix:"0#@"}});console.log("A114_PREFIX_SETTER",JSON.stringify({syntheticPrefixAccepted:"0#@",sequenceResetRequested:false,existingSequence:42,referenceRestartModeImplemented:false}))});
it("renders the actual year-list Web page without running its form action",async()=>{const html=renderToStaticMarkup(await Page());expect(html).toContain("Financial Years");expect(html).toContain("Verify &amp; close Financial Year");expect(html).not.toContain("Restart Transaction Numbers");expect(html).not.toContain("Start Fresh");expect(m.update).not.toHaveBeenCalled();expect(m.permission).toHaveBeenCalledWith("ACCOUNT_PERIOD_LOCK");console.log("A114_WEB_FLOW",JSON.stringify({yearList:true,referenceModeChooser:false,backupStep:false,closeExecuted:false}))});
