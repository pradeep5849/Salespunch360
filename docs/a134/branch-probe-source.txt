import {beforeEach,expect,it,vi} from 'vitest';
import {readFileSync} from 'node:fs';
const m=vi.hoisted(()=>({owner:vi.fn(),create:vi.fn(),find:vi.fn(),cookie:vi.fn()}));
vi.mock('@/lib/company/owner-authorization',()=>({requirePrimaryOwner:m.owner}));
vi.mock('@/lib/db',()=>({db:{branch:{create:m.create,findMany:m.find}}}));
vi.mock('next/headers',()=>({cookies:async()=>({get:m.cookie})}));
import {createBranch,branchDetails,listBranches} from '@/lib/branches/management';
import {resolveAccountBranchContext} from '@/lib/account/branch-context';
import {operationalBranchContext,resolveWriteBranch} from '@/lib/branches/operational-scope';
const data={name:' First location ',code:' fl '};
const actor={id:'owner',companyId:'company-A',accountRole:'ACCOUNT_ADMIN' as const,branchAccessScope:'ALL_BRANCHES' as const,branchIds:[]};
beforeEach(()=>{vi.clearAllMocks();m.owner.mockResolvedValue({actor});m.find.mockResolvedValue([]);m.create.mockImplementation(({data}:any)=>({id:'synthetic-branch',...data}))});
it('uses mutation owner authorization and authenticated company on create',async()=>{const result=await createBranch(data);expect(m.owner).toHaveBeenCalledWith(true);expect(result).toMatchObject({companyId:'company-A',name:'First location',code:'FL',isPrimary:false})});
it('denies unauthorized create before database mutation',async()=>{m.owner.mockRejectedValue(new Error('Not authorized'));await expect(createBranch(data)).rejects.toThrow('Not authorized');expect(m.create).not.toHaveBeenCalled()});
it('list is owner-authorized and company scoped',async()=>{await listBranches();expect(m.owner).toHaveBeenCalledOnce();expect(m.find.mock.calls[0][0].where).toEqual({companyId:'company-A'})});
it('rejects injected company identity in strict schema',()=>{expect(()=>branchDetails.parse({...data,companyId:'foreign'})).toThrow()});
it('first create performs no existing-data choice or transactional reassignment',async()=>{await createBranch(data);expect(m.create).toHaveBeenCalledOnce();expect(m.find).not.toHaveBeenCalled();expect(m.create.mock.calls[0][0].data).not.toHaveProperty('existingDataChoice')});
it('schema cannot express either agreed first-branch choice',()=>{for(const existingDataChoice of ['MOVE_TO_FIRST_BRANCH','KEEP_COMPANY_LEVEL'])expect(()=>branchDetails.parse({...data,existingDataChoice})).toThrow()});
it('Account zero-branch company cannot resolve normal company context',async()=>{await expect(resolveAccountBranchContext(actor)).rejects.toThrow();expect(m.find).toHaveBeenCalled()});
it('Sales server-loaded zero-branch snapshot is forbidden',async()=>{await expect(operationalBranchContext(actor)).rejects.toThrow('BRANCH_FORBIDDEN')});
it('Sales write resolver cannot select company-level zero-branch mode',()=>{expect(()=>resolveWriteBranch([])).toThrow('BRANCH_FORBIDDEN')});
it('registration still automatically creates Head Office',()=>{const s=readFileSync('src/lib/auth/registration.ts','utf8');expect(s).toContain('await tx.branch.create(');expect(s).toContain('name: "Head Office", code: "HO", isPrimary: true')});
it('first-create form has no move-versus-company-level selection',()=>{const s=readFileSync('src/components/workspace/branch-management-form.tsx','utf8');expect(s).not.toContain('existingDataChoice');expect(s).not.toContain('type="radio"')});
it('general settings branch entry is unconditional on branch creation state',()=>{const s=readFileSync('src/components/account/general-settings-form.tsx','utf8');expect(s).toContain('<Link className="general-settings-row navigates" href="/workspace/account/settings/multi-branch">');expect(s).not.toContain('branchesEnabled')});
