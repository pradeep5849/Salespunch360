import {beforeEach,expect,it,vi} from 'vitest';
import {readFileSync} from 'node:fs';
const m=vi.hoisted(()=>({company:vi.fn(),target:vi.fn(),subscriptions:vi.fn(),count:vi.fn(),update:vi.fn(),revoke:vi.fn(),branches:vi.fn()}));
vi.mock('@/lib/db',()=>({db:{company:{findUnique:m.company},$transaction:async(fn:any)=>fn({$queryRaw:vi.fn(async()=>[]),user:{findFirst:m.target,count:m.count,update:m.update},companySubscription:{findMany:m.subscriptions},userBranchAccess:{deleteMany:m.branches,createMany:vi.fn()},branch:{count:vi.fn(async()=>1)}})}}));
vi.mock('@/lib/auth/session-generation',()=>({revokeUserAuthenticationWithLock:m.revoke}));
import {editAccountUserForActor} from '@/lib/users/product-user-management';
const actor={id:'admin',companyId:'company-A',role:'ACCOUNT_USER',isActive:true,accountRole:'ACCOUNT_ADMIN',accountAccessActive:true,salesRole:null,salesAccessActive:false};
const input={userId:'11111111-1111-4111-8111-111111111111',accountRole:'ACCOUNTANT',accountAccessActive:true,branchAccessScope:'ALL_BRANCHES',branchIds:[]};
const target={id:input.userId,companyId:'company-A',isActive:true,salesRole:null,accountRole:'DATA_ENTRY',salesAccessActive:false,accountAccessActive:true,branchAccessScope:'ALL_BRANCHES',branchAccesses:[]};
beforeEach(()=>{vi.clearAllMocks();m.company.mockResolvedValue({id:'company-A',productEdition:'SALESPUNCH360_ACCOUNT',teamStructure:'SALES_ONLY'});m.target.mockResolvedValue(target);m.subscriptions.mockResolvedValue([{adminSeats:1,accountPackages:1,sourceOrder:{provider:'ACCOUNT_PACKAGE'}}]);m.count.mockResolvedValue(1)});
it('reproduces second role seat permitted by one paid package plus unconditional base package',async()=>{await editAccountUserForActor(actor as any,input);expect(m.update).toHaveBeenCalled();expect(m.count.mock.calls[0][0].where).toMatchObject({accountRole:'ACCOUNTANT',id:{not:input.userId}});expect(m.revoke).toHaveBeenCalledWith(expect.anything(),input.userId)});
it('with two occupied role seats the same calculation rejects',async()=>{m.count.mockResolvedValue(2);await expect(editAccountUserForActor(actor as any,input)).rejects.toThrow('ACCOUNT_SEAT_LIMIT');expect(m.update).not.toHaveBeenCalled()});
it('protects Primary Admin target from role and access changes',async()=>{m.target.mockResolvedValue({...target,salesRole:'PRIMARY_ADMIN'});await expect(editAccountUserForActor(actor as any,input)).rejects.toThrow('PRIMARY_PROTECTED');expect(m.update).not.toHaveBeenCalled()});
it('rejects globally inactive identity activation',async()=>{m.target.mockResolvedValue({...target,isActive:false});await expect(editAccountUserForActor(actor as any,input)).rejects.toThrow('IDENTITY_INACTIVE')});
it('shared Sales branch changes are forbidden',async()=>{m.target.mockResolvedValue({...target,salesRole:'SALES',branchAccessScope:'SELECTED_BRANCHES',branchAccesses:[{branchId:'other'}]});await expect(editAccountUserForActor(actor as any,input)).rejects.toThrow('SALES_BRANCH_AUTHORITY_REQUIRED');expect(m.update).not.toHaveBeenCalled()});
it('shared identity keeps Sales branch authority when Account role changes',async()=>{m.target.mockResolvedValue({...target,salesRole:'SALES'});await editAccountUserForActor(actor as any,input);const d=m.update.mock.calls[0][0].data;expect(d).not.toHaveProperty('salesRole');expect(d).not.toHaveProperty('branchAccessScope');expect(m.branches).not.toHaveBeenCalled()});
it('native UserView offers toggles but no existing-user role/branch editor',()=>{const s=readFileSync('android/app/src/main/java/com/salespunch360/mobile/ui/account/admin/AccountAdministrationScreen.kt','utf8');const cards=s.slice(s.indexOf('Text("Effective roles and access")'),s.indexOf('private val accountSettingsDestinations'));expect(cards).toContain('Deactivate Account access');expect(cards).not.toContain('OutlinedTextField');expect(cards).not.toContain('DropdownMenu');expect(cards).not.toContain('BranchAccessEditor')});
