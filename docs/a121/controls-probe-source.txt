import { beforeEach, expect, it, vi } from 'vitest';
const m=vi.hoisted(()=>({states:[] as any[],index:0,effects:[] as any[],writes:[] as any[],start:vi.fn(),end:vi.fn(),upload:vi.fn(),refresh:vi.fn(),position:vi.fn(),watch:vi.fn(),clear:vi.fn()}));
vi.mock('react',async original=>({...await original<typeof import('react')>(),useState:(initial:any)=>{const i=m.index++;if(!(i in m.states))m.states[i]=initial;return[m.states[i],(value:any)=>{m.states[i]=value;m.writes.push(value)}]},useRef:()=>({current:0}),useEffect:(effect:any)=>m.effects.push(effect),useTransition:()=>[false,(fn:any)=>fn()]}));
vi.mock('next/navigation',()=>({useRouter:()=>({refresh:m.refresh})}));
vi.mock('@/app/actions/attendance',()=>({startAttendanceAction:m.start,endAttendanceAction:m.end,uploadLocationPointAction:m.upload}));
import { AttendanceControls } from '@/app/workspace/attendance/attendance-controls';
function render(open=false,gps=false){m.index=0;m.effects=[];return AttendanceControls({initialOpen:open,startedAt:'2026-11-11T02:00:00Z',gpsEnabled:gps,attendanceEnabled:true,pointCount:0})}
function button(node:any):any{if(!node)return;if(Array.isArray(node)){for(const child of node){const found=button(child);if(found)return found}}else if(node.type==='button')return node;else return button(node.props?.children)}
beforeEach(()=>{vi.clearAllMocks();m.states=[];m.index=0;m.effects=[];m.writes=[];m.start.mockResolvedValue({ok:true});m.end.mockResolvedValue({ok:true});m.upload.mockResolvedValue({ok:true});m.position.mockImplementation((ok:any)=>ok({coords:{latitude:0,longitude:0,accuracy:5},timestamp:Date.now()}));m.watch.mockReturnValue(1);vi.stubGlobal('navigator',{geolocation:{getCurrentPosition:m.position,watchPosition:m.watch,clearWatch:m.clear}})});
it('reproduces confirmed start refresh with no explicit success message',async()=>{await button(render()).props.onClick();expect(m.start).toHaveBeenCalledOnce();expect(m.refresh).toHaveBeenCalledOnce();expect(m.writes.some(x=>typeof x==='string'&&/success|started/i.test(x))).toBe(false)});
it('reproduces confirmed stop refresh with no explicit success message',async()=>{await button(render(true)).props.onClick();expect(m.end).toHaveBeenCalledOnce();expect(m.refresh).toHaveBeenCalledOnce();expect(m.writes.some(x=>typeof x==='string'&&/success|stopped/i.test(x))).toBe(false)});
it('shows server rejection and does not refresh as success',async()=>{m.end.mockResolvedValue({ok:false,error:'Server rejected'});await button(render(true)).props.onClick();expect(m.writes).toContain('Server rejected');expect(m.refresh).not.toHaveBeenCalled()});
it('GPS denial prevents stop submission before contacting server',async()=>{m.position.mockImplementation((_ok:any,error:any)=>error({code:1}));await button(render(true,true)).props.onClick();expect(m.end).not.toHaveBeenCalled();expect(m.writes.some(x=>typeof x==='string'&&x.includes('permission was denied'))).toBe(true)});
it('reproduces unhandled submission transport rejection',async()=>{m.start.mockRejectedValue(new Error('transport disconnected'));await expect(button(render()).props.onClick()).rejects.toThrow('transport disconnected');expect(m.refresh).not.toHaveBeenCalled();expect(m.writes).not.toContain('transport disconnected')});
it('one rejected GPS result stops watching and has no automatic retry',async()=>{m.upload.mockResolvedValue({ok:false,error:'temporary outage'});render(true,true);m.effects[0]();const position=m.watch.mock.calls[0][0];position({coords:{latitude:0,longitude:0,accuracy:5},timestamp:Date.now()});await Promise.resolve();expect(m.writes).toContain(true);const cleanup=m.effects[0];void cleanup;m.watch.mockClear();render(true,true);m.effects[0]();expect(m.watch).not.toHaveBeenCalled()});
