import { beforeEach, expect, it, vi } from 'vitest';
import { Prisma } from '@prisma/client';
const m=vi.hoisted(()=>({permission:vi.fn(),db:{} as Record<string,unknown>}));
vi.mock('@/lib/auth/authorization',()=>({AuthorizationError:class extends Error{},requirePermission:m.permission,requirePermissionForMutation:vi.fn()}));
vi.mock('@/lib/db',()=>({db:m.db}));
import { verifyAccountDataForActor, verifyAccountData } from '@/lib/account/utilities';
const D=(n:number)=>new Prisma.Decimal(n);
function fixture(){
 const arrays:Record<string,unknown[]>={journalEntry:[],commercialDocument:[],ledgerAccount:[],moneyAccount:[],stockMovement:[],project:[],asset:[],assetAssignmentHistory:[],user:[],branch:[],customer:[]};
 const reads:unknown[]=[];
 const client=Object.fromEntries(Object.keys(arrays).map(k=>[k,{findMany:vi.fn(async(q:unknown)=>{reads.push({model:k,q});return arrays[k]}),groupBy:vi.fn(async(q:unknown)=>{reads.push({model:k,q});return arrays[k]})}]));
 return {arrays,client,reads};
}
beforeEach(()=>{vi.clearAllMocks();for(const k of Object.keys(m.db))delete m.db[k]});
it('baseline runs ten checks using reads only and company-filtered queries',async()=>{const f=fixture();const r=await verifyAccountDataForActor({id:'reader',companyId:'company'},f.client as never);expect(r).toHaveLength(10);expect(r.every(x=>x.status==='PASS')).toBe(true);for(const r of f.reads)expect((r as {q:{where:unknown}}).q.where).toMatchObject({companyId:'company'})});
it('detects imbalanced journals and inconsistent tax component sums',async()=>{const f=fixture();f.arrays.journalEntry=[{journalNumber:'JV-1',lines:[{debit:D(100),credit:D(0)}]}];f.arrays.commercialDocument=[{documentNumber:'INV-1',balanceDue:D(100),grandTotal:D(100),payableAmount:D(100),cgstTotal:D(9),sgstTotal:D(9),igstTotal:D(0),cessTotal:D(0),taxTotal:D(0)}];const r=await verifyAccountDataForActor({id:'reader',companyId:'company'},f.client as never);expect(r.find(x=>x.check==='Journal entries balance')?.status).toBe('ERROR');expect(r.find(x=>x.check==='Stored tax components reconcile')?.status).toBe('ERROR')});
it('reproduces false pass when transfer legs have mismatched quantities and products',async()=>{const f=fixture();f.arrays.stockMovement=[{id:'in',sourceId:'transfer',movementType:'TRANSFER_IN',quantity:D(1),warehouseId:'w1',productId:'p1'},{id:'out',sourceId:'transfer',movementType:'TRANSFER_OUT',quantity:D(99),warehouseId:'w2',productId:'p2'}];const r=await verifyAccountDataForActor({id:'reader',companyId:'company'},f.client as never);expect(r.find(x=>x.check==='Inventory transfers are paired')?.status).toBe('PASS')});
it('reproduces company-wide web samples for a selected-branch report reader',async()=>{const f=fixture();Object.assign(m.db,f.client);m.permission.mockResolvedValue({id:'reader',companyId:'company',accountRole:'ACCOUNTANT',branchAccessScope:'SELECTED_BRANCHES',branchIds:['allowed-branch']});f.arrays.journalEntry=[{journalNumber:'OTHER-BRANCH-JV',lines:[{debit:D(5),credit:D(0)}]}];const r=await verifyAccountData();expect(m.permission).toHaveBeenCalledWith('ACCOUNT_REPORTS');expect(r.find(x=>x.check==='Journal entries balance')?.samples).toEqual(['OTHER-BRANCH-JV']);expect((f.client.journalEntry.findMany.mock.calls[0][0] as {where:unknown}).where).toEqual({companyId:'company'})});
it('reproduces all-pass result when no canonical system ledgers exist',async()=>{const f=fixture();const r=await verifyAccountDataForActor({id:'reader',companyId:'company'},f.client as never);expect(r.find(x=>x.check==='System ledgers are unique')?.status).toBe('PASS')});
it('limits displayed samples to five while retaining the affected count',async()=>{const f=fixture();f.arrays.journalEntry=Array.from({length:7},(_,i)=>({journalNumber:`JV-${i}`,lines:[{debit:D(1),credit:D(0)}]}));const r=await verifyAccountDataForActor({id:'reader',companyId:'company'},f.client as never);expect(r[0].affected).toBe(7);expect(r[0].samples).toHaveLength(5)});
it('confirms actual report policy allows a selected-branch Accountant',async()=>{const {canUsePermission}=await import('@/lib/auth/permissions');expect(canUsePermission({id:'reader',companyId:'company',role:'ACCOUNT_USER',isActive:true,accountAccessActive:true,salesAccessActive:false,accountRole:'ACCOUNTANT',salesRole:null,managerType:null,branchAccessScope:'SELECTED_BRANCHES',branchIds:['allowed-branch']} as never,'SALESPUNCH360_ACCOUNT','ACCOUNT_REPORTS')).toBe(true)});
