import {beforeEach,expect,it,vi} from 'vitest';
import {readFileSync} from 'node:fs';
const m=vi.hoisted(()=>({owner:vi.fn(),process:vi.fn(),put:vi.fn(),update:vi.fn(),profile:vi.fn(),refresh:vi.fn()}));
vi.mock('@/lib/company/owner-authorization',()=>({requirePrimaryOwner:m.owner}));
vi.mock('@/lib/company/profile',()=>({updateCompanyProfile:m.profile}));
vi.mock('@/lib/company/logo',()=>({companyLogoKey:(id:string)=>`Logo/${id}.webp`,processCompanyLogo:m.process}));
vi.mock('@/lib/storage',()=>({privateStorage:()=>({put:m.put})}));
vi.mock('@/lib/db',()=>({db:{company:{update:m.update}}}));
vi.mock('next/cache',()=>({revalidatePath:m.refresh}));
import {replaceCompanyLogo,saveCompanyProfile} from '@/app/actions/company-profile';
const form=()=>{const f=new FormData();f.set('companyLogo',new File(['synthetic'],'logo.webp',{type:'image/webp'}));return f};
beforeEach(()=>{vi.clearAllMocks();m.owner.mockResolvedValue({actor:{id:'owner',companyId:'company-A'}});m.process.mockResolvedValue(Buffer.from('synthetic-new-logo'));m.put.mockResolvedValue(undefined);m.update.mockResolvedValue({});m.profile.mockResolvedValue({})});
it('logo upload derives owner company key and only then saves pointer',async()=>{expect(await replaceCompanyLogo({},form())).toHaveProperty('success');expect(m.owner).toHaveBeenCalledWith(true);expect(m.put).toHaveBeenCalledWith('Logo/company-A.webp',expect.any(Buffer));expect(m.update).toHaveBeenCalledWith({where:{id:'company-A'},data:{logoObjectKey:'Logo/company-A.webp'}})});
it('logo database failure follows already completed overwrite and returns error without refresh',async()=>{m.update.mockRejectedValueOnce(new Error('database failed'));expect(await replaceCompanyLogo({},form())).toHaveProperty('error');expect(m.put).toHaveBeenCalledOnce();expect(m.refresh).not.toHaveBeenCalled();expect(m.put.mock.invocationCallOrder[0]).toBeLessThan(m.update.mock.invocationCallOrder[0])});
it('logo authorization failure precedes processing and storage',async()=>{m.owner.mockRejectedValueOnce(new Error('Not authorized'));expect(await replaceCompanyLogo({},form())).toHaveProperty('error');expect(m.process).not.toHaveBeenCalled();expect(m.put).not.toHaveBeenCalled()});
it('storage failure does not update logo pointer',async()=>{m.put.mockRejectedValueOnce(new Error('storage failed'));expect(await replaceCompanyLogo({},form())).toHaveProperty('error');expect(m.update).not.toHaveBeenCalled()});
it('web maps manager-seat conflict to actionable text',async()=>{m.profile.mockRejectedValueOnce(new Error('TEAM_STRUCTURE_CONFLICT'));const f=new FormData();expect(await saveCompanyProfile({},f)).toMatchObject({error:expect.stringContaining('Manager seats')})});
it('Android omits equivalent manager conflict recovery guidance',()=>{const s=readFileSync('android/app/src/main/java/com/salespunch360/mobile/CompanyProfileViewModel.kt','utf8');expect(s).not.toContain('TEAM_STRUCTURE_CONFLICT');expect(s).toContain('Company details couldn\'t be updated.')});
it('mobile company reads require settings authority rather than shared COMPANY_VIEW',()=>{const s=readFileSync('src/lib/mobile/company.ts','utf8');expect(s).toContain('mobileCan(principal, "SALES_SETTINGS")');const start=s.indexOf('export async function mobileCompanyContext');expect(s.slice(start,start+220)).toContain('admin(principal)')});
it('web profile exposes internal primary branch label independently of branch activation',()=>{const s=readFileSync('src/app/workspace/company-profile/profile-form.tsx','utf8');expect(s).toContain('<h2>Primary Branch</h2>');expect(s).toContain('profile.primaryBranch?.name??"Head Office"');expect(s).not.toContain('branchesEnabled')});
it('mobile logo uses the same write-before-pointer pattern',()=>{const s=readFileSync('src/lib/mobile/company.ts','utf8');expect(s.indexOf('await privateStorage().put(key, logo)')).toBeLessThan(s.indexOf('data: { logoObjectKey: key }'));expect(s).toContain('companyLogoKey(companyId)')});
