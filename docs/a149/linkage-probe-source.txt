import {beforeEach,expect,it,vi} from 'vitest';
import {readFileSync} from 'node:fs';
const m=vi.hoisted(()=>({actor:vi.fn(),find:vi.fn(),insert:vi.fn(),query:vi.fn(),paid:vi.fn()}));
vi.mock('@/lib/auth/authorization',()=>({requireSalesWorkspace:m.actor,requireSalesWorkspaceForMutation:m.actor}));
vi.mock('@/lib/billing/telecaller',()=>({assertActiveTelecallerEntitlement:m.paid}));
vi.mock('@/lib/db',()=>({db:{lead:{findFirst:m.find},$queryRaw:m.query,$transaction:async(fn:any)=>fn({$executeRaw:m.insert})}}));
import {recordLeadCall} from '@/lib/telecalling/service';
beforeEach(()=>{vi.clearAllMocks();m.actor.mockResolvedValue({id:'tele',companyId:'company-A',salesRole:'SALES',designation:'Telecaller'});m.find.mockResolvedValue({id:'lead',companyId:'company-A',assignedUserId:'sales'});m.insert.mockResolvedValue(1)});
it('CALL_BACK inserts only call history without creating a shared follow-up task',async()=>{await recordLeadCall({leadId:'lead',result:'CALL_BACK',nextCallbackAt:new Date(Date.now()+86400000)});expect(m.insert).toHaveBeenCalledOnce();const query=m.insert.mock.calls[0][0];expect(query.strings.join('')).toContain('INSERT INTO "lead_calls"');expect(query.strings.join('')).not.toContain('follow_up_tasks')});
it('rejects missing callback date before saving',async()=>{await expect(recordLeadCall({leadId:'lead',result:'CALL_BACK'})).rejects.toThrow('CALLBACK_DATE_REQUIRED');expect(m.insert).not.toHaveBeenCalled()});
it('rejects past callback date before saving',async()=>{await expect(recordLeadCall({leadId:'lead',result:'CALL_BACK',nextCallbackAt:new Date('2020-01-01')})).rejects.toThrow('CALLBACK_DATE_MUST_BE_FUTURE');expect(m.insert).not.toHaveBeenCalled()});
it('normal lead detail omits phone and email displayed on Android',()=>{const web=readFileSync('src/app/workspace/leads/[id]/page.tsx','utf8');const native=readFileSync('android/app/src/main/java/com/salespunch360/mobile/ui/LeadsScreen.kt','utf8');expect(web).not.toContain('lead.phone');expect(web).not.toContain('lead.email');expect(native).toContain('Phone: $it');expect(native).toContain('Email: $it')});
it('web lead history is capped at 100 without page controls while native supports follow-up load more',()=>{const loader=readFileSync('src/lib/follow-up-tasks/lead-list.ts','utf8');const web=readFileSync('src/app/workspace/leads/[id]/page.tsx','utf8');const native=readFileSync('android/app/src/main/java/com/salespunch360/mobile/ui/LeadsScreen.kt','utf8');expect(loader).toContain('take:100');expect(web).not.toContain('Load more');expect(web).not.toContain('page=');expect(native).toContain('Load more follow-ups')});
it('explicit Telecaller assignments are merged into the Telecalling queue',()=>{const web=readFileSync('src/app/workspace/telecalling/page.tsx','utf8');expect(web).toContain('splitCallbackQueue([...callbacks,...assignedCalls])');const linked=readFileSync('src/lib/follow-up-tasks/telecaller-assignment.ts','utf8');expect(linked).toContain('FROM "follow_up_tasks"');expect(linked).toContain('status:\"COMPLETED\"')});
